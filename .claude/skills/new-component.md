# Skill: new-component

Create a new React component — either a shared UI component or a module-scoped component.

## Usage

```
/new-component <location>/<ComponentName>
```

Examples:
- `/new-component auth/LoginForm`          → `src/modules/auth/components/LoginForm.tsx`
- `/new-component shared/layout/PageHeader` → `src/components/layout/PageHeader.tsx`
- `/new-component shared/ui/StatusBadge`   → `src/components/ui/StatusBadge.tsx`

---

## Location rules

| Prefix | Output path | Use when |
|---|---|---|
| `<module>/` | `src/modules/<module>/components/` | Used only inside one feature module |
| `shared/layout/` | `src/components/layout/` | Layout chrome used across modules |
| `shared/form/` | `src/components/form/` | Reusable form building blocks |
| `shared/ui/` | `src/components/ui/` | Generic display components (NOT shadcn primitives) |

> `src/components/ui/` shadcn primitives are generated by `npx shadcn add` — never hand-edit them.
> Custom UI components that wrap or compose shadcn go in `src/components/ui/` with a distinct name.

## Template — module component

```tsx
// src/modules/<module>/components/<ComponentName>.tsx
import { cn } from "@/lib/utils"
import type { <RelatedType> } from "../types/<module>.types"

interface <ComponentName>Props {
  // define props
  className?: string
}

export function <ComponentName>({ className }: <ComponentName>Props) {
  return (
    <div className={cn("", className)}>
      {/* content */}
    </div>
  )
}
```

## Template — shared component

```tsx
// src/components/<type>/<ComponentName>.tsx
import { cn } from "@/lib/utils"

interface <ComponentName>Props {
  // define props
```

## Full example — `src/modules/user/components/UserAvatar.tsx`

```tsx
import { cn } from "@/lib/utils"
import type { User } from "../types/user.types"

interface UserAvatarProps {
  user: User
  size?: "sm" | "md" | "lg"
  className?: string
}

export function UserAvatar({ user, size = "md", className }: UserAvatarProps) {
  const sizeClass = {
    sm: "size-6 text-xs",
    md: "size-9 text-sm",
    lg: "size-14 text-lg",
  }[size]

  return (
    <div
      className={cn(
        "flex items-center justify-center rounded-full bg-muted font-medium text-muted-foreground",
        sizeClass,
        className
      )}
      aria-label={user.name}
    >
      {user.name.charAt(0).toUpperCase()}
    </div>
  )
}
```

## Full example — `src/components/layout/PageHeader.tsx`

```tsx
import { cn } from "@/lib/utils"

interface PageHeaderProps {
  title: string
  description?: string
  action?: React.ReactNode
  className?: string
}

export function PageHeader({ title, description, action, className }: PageHeaderProps) {
  return (
    <div className={cn("flex items-start justify-between gap-4 pb-6", className)}>
      <div className="space-y-1">
        <h1 className="text-xl font-semibold tracking-tight text-foreground">{title}</h1>
        {description && (
          <p className="text-sm text-muted-foreground">{description}</p>
        )}
      </div>
      {action && <div className="shrink-0">{action}</div>}
    </div>
  )
}
```

## Conventions

- Named export: `export function <ComponentName>() {}`
- No `default` export
- No `any` type — define a `<ComponentName>Props` interface for all props
- Always accept an optional `className?: string` prop on components that render a root element
- Use `cn()` from `@/lib/utils` to merge classNames
- Tailwind utility classes only — no inline styles
- Import types from the module's `types/` folder, not from the component's own file
- Icons: `import { XIcon } from "lucide-react"` — use `data-icon="inline-end"` when beside text, or `className="size-4 text-muted-foreground"` standalone
